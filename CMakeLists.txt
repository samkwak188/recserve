cmake_minimum_required(VERSION 3.24)
project(recserve LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(RECSERVE_SANITIZER "none" CACHE STRING "address|undefined|thread|none")
option(RECSERVE_WITH_RDKAFKA "Link librdkafka if found" OFF)
option(RECSERVE_WERROR "Treat warnings as errors" ON)
option(RECSERVE_WITH_AVX2 "Require AVX2 and FMA on the deployment CPU" OFF)
option(RECSERVE_WITH_CUDA "Build the CUDA exact retrieval backend" OFF)

if(MSVC)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>)
  if(RECSERVE_WERROR)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/WX>)
  endif()
else()
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Wall> $<$<COMPILE_LANGUAGE:CXX>:-Wextra>)
  if(RECSERVE_WERROR)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Werror>)
  endif()
endif()

if(RECSERVE_WITH_AVX2)
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    message(FATAL_ERROR "RECSERVE_WITH_AVX2 requires an x86-64 target")
  endif()
  if(MSVC)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/arch:AVX2>)
  else()
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-mavx2> $<$<COMPILE_LANGUAGE:CXX>:-mfma>)
  endif()
endif()

if(NOT RECSERVE_SANITIZER STREQUAL "none" AND NOT MSVC)
  add_compile_options(-fsanitize=${RECSERVE_SANITIZER} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${RECSERVE_SANITIZER})
endif()

add_library(recserve_net STATIC src/net.cpp src/kafka_source.cpp)
target_include_directories(recserve_net PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

if(RECSERVE_WITH_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(recserve_gpu STATIC src/gpu_scorer.cu)
  set_target_properties(recserve_gpu PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(recserve_gpu PRIVATE CUDA::cudart CUDA::cublas)
  target_compile_definitions(recserve_gpu PUBLIC RECSERVE_HAS_CUDA=1)
else()
  add_library(recserve_gpu STATIC src/gpu_stub.cpp)
endif()
target_include_directories(recserve_gpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

function(recserve_app name src)
  add_executable(${name} ${src})
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(${name} PRIVATE recserve_net recserve_gpu)
  if(WIN32)
    target_link_libraries(${name} PRIVATE ws2_32)
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(${name} PRIVATE Threads::Threads)
  endif()
endfunction()

recserve_app(recserve_tests tests/test_all.cpp)
recserve_app(recserve_server apps/server.cpp)
recserve_app(recserve_loadgen apps/loadgen.cpp)
recserve_app(recserve_quality apps/quality.cpp)
recserve_app(recserve_bench apps/bench.cpp)
recserve_app(recserve_nearline apps/nearline.cpp)
recserve_app(recserve_soak apps/soak.cpp)
recserve_app(recserve_rankd apps/rankd.cpp)
recserve_app(recserve_diagnose apps/diagnose.cpp)
recserve_app(recserve_fixture apps/fixture.cpp)
recserve_app(recserve_eval apps/eval.cpp)
recserve_app(recserve_shard apps/shard.cpp)
recserve_app(recserve_gpu_bench apps/gpu_bench.cpp)
recserve_app(recserve_gpu_tests tests/test_gpu.cpp)
recserve_app(recserve_batcher_tests tests/test_batcher.cpp)

enable_testing()
add_test(NAME recserve_gpu COMMAND recserve_gpu_tests)
add_test(NAME recserve_batcher COMMAND recserve_batcher_tests)
# Run from the source tree so fixtures under data/ resolve the same way they
# do for a developer running the binary by hand.
add_test(NAME recserve_unit COMMAND recserve_tests
         WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME recserve_service COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_service.py --build $<TARGET_FILE_DIR:recserve_server>)
  add_test(NAME recserve_artifacts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_artifacts.py --build $<TARGET_FILE_DIR:recserve_bench>)
  if(RECSERVE_WITH_CUDA)
    add_test(NAME recserve_service_cuda COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_service.py --build $<TARGET_FILE_DIR:recserve_server> --backend cuda)
  endif()
endif()

if(RECSERVE_WITH_RDKAFKA)
  find_library(RDKAFKA_CPP NAMES rdkafka++ rdkafkacpp)
  find_library(RDKAFKA NAMES rdkafka)
  if(RDKAFKA_CPP AND RDKAFKA)
    message(STATUS "librdkafka found: ${RDKAFKA_CPP}")
    target_compile_definitions(recserve_net PUBLIC RECSERVE_HAS_RDKAFKA=1)
    target_link_libraries(recserve_net PUBLIC ${RDKAFKA_CPP} ${RDKAFKA})
  else()
    message(WARNING "RECSERVE_WITH_RDKAFKA=ON but librdkafka was not found; "
                    "the Kafka source will report itself unavailable at runtime")
  endif()
endif()
